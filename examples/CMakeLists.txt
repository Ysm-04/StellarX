function(stellarx_add_example target directory)
    add_executable(${target} "${directory}/main.cpp")
    target_link_libraries(${target} PRIVATE StellarX)
    if(MSVC)
        target_compile_options(${target} PRIVATE ${STELLARX_MSVC_TEXT_OPTIONS})
    endif()
endfunction()

add_executable(stellarx-basic-window WIN32 "basic-window/main.cpp")
target_link_libraries(stellarx-basic-window PRIVATE StellarX)
if(MSVC)
    target_compile_options(stellarx-basic-window PRIVATE ${STELLARX_MSVC_TEXT_OPTIONS})
endif()
